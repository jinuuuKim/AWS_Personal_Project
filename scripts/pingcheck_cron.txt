###서울###
cat <<EOF > /root/pingcheck.sh
#!/bin/bash
HOST_TO_CHECK=10.2.1.100

if /usr/bin/ping -qc 20 -W 1 \$HOST_TO_CHECK >/dev/null; then
    echo "IDC-DB \$HOST_TO_CHECK is up"
    systemctl start httpd
else
    echo "IDC-DB \$HOST_TO_CHECK is down"
    systemctl stop httpd
fi
EOF

chmod +x /root/pingcheck.sh
echo "* * * * * /root/pingcheck.sh >> /root/db_check.log 2>&1" > /var/spool/cron/root
chmod 600 /var/spool/cron/root
